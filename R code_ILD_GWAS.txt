library(missRanger)
library(ranger)

df_train <- read.csv("C:/Users/line1/Desktop/台中榮總/ILD_GWAS/Data/ILD_df_train.csv", header = TRUE)
df_test  <- read.csv("C:/Users/line1/Desktop/台中榮總/ILD_GWAS/Data/ILD_df_test.csv",  header = TRUE)

col_names <- setdiff(names(df_train), c("Outcome", "C_ID"))

df_train_miss <- df_train[, col_names]
df_test_miss  <- df_test[, col_names]

set.seed(20261010)
train_result <- missRanger(
  data = df_train_miss,
  num.trees = 50,
  maxiter = 5,
  pmm.k = 3,
  # keep_models = TRUE,
  verbose = 0
)

test_result <- missRanger:::predict.missRanger(train_result, newdata = df_test_miss)
test_result <- as.data.frame(test_result)
train_clean <- as.data.frame(train_result)

residual_cols <- names(which(colSums(is.na(test_result)) > 0))

if (length(residual_cols) > 0) {

  for (col in residual_cols) {
    
    na_idx <- which(is.na(test_result[[col]]))
    
    predictor_cols <- setdiff(col_names, residual_cols)
    
    rf_formula <- as.formula(paste(col, "~ ."))
    
    set.seed(20261010)
    rf_model <- ranger(
      formula = rf_formula,
      data = train_clean[, c(col, predictor_cols)],
      num.trees = 50,
      seed = 20261010
    )
    
    pred_vals <- predict(rf_model, data = test_result[na_idx, predictor_cols])$predictions
    
    test_result[na_idx, col] <- pred_vals
  }
}

df_train[, col_names] <- train_clean
df_test[, col_names]  <- test_result
